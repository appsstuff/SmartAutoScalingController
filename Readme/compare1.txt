def retrain_all_models():
    print("🔄 Starting retraining process using VictoriaMetrics data...")

    for svc in AUTO_SCALE_SERVICES:
        try:
            pod_name = svc["pod_name"]
            namespace = svc.get("namespace", "default")
            seq_length = svc.get("seq_length", 10)

            print(f"\n📡 Fetching data for {pod_name} ({namespace})")
            cpu_values = fetch_historical_data(pod_name, namespace, days=10)

            if len(cpu_values) < seq_length:
                print(f"[SKIP] Not enough CPU data for {pod_name}")
                continue

            # Generate synthetic features
            synthetic_features = []
            for v in cpu_values[-seq_length:]:
                synthetic_features.append(np.array([
                    datetime.now().hour,
                    datetime.now().weekday(),
                    v * 0.9,
                    v * 0.7,
                    v * 0.8,
                    np.random.uniform(200, 300),
                    np.random.uniform(5, 20)
                ]))
            X_train = np.array(synthetic_features)

            # Labels
            y_train = fetch_logged_decisions(pod_name, namespace, days=10)
            if len(y_train) == 0:
                print(f"[WARN] No decisions found — using synthetic labels for {pod_name}")
                y_train = np.random.randint(0, 3, size=len(X_train))

            # Normalize
            scaler = StandardScaler()
            X_scaled = scaler.fit_transform(X_train)
            joblib.dump(scaler, f"models/scaler_model_{pod_name}.pkl")

            # Create sequence
            X_seq = create_sequence(X_scaled, seq_length)
            if len(X_seq) == 0:
                print(f"[SKIP] Not enough sequence data for {pod_name}")
                continue
            y_seq = y_train[-len(X_seq):]

            # Retrain
            gpr = GaussianProcessRegressor()
            gpr.fit(X_scaled, y_train)
            joblib.dump(gpr, f"models/gpr_model_{pod_name}.pkl")

            clf = XGBClassifier(use_label_encoder=False, eval_metric='mlogloss')
            clf.fit(X_scaled, y_train)
            clf.save_model(f"models/xgb_model_{pod_name}.json")

            model = Sequential([
                LSTM(64, input_shape=(X_seq.shape[1], X_seq.shape[2]), return_sequences=False),
                Dropout(0.3),
                Dense(32, activation='relu'),
                Dense(1, activation='sigmoid')
            ])
            model.compile(optimizer='adam', loss='binary_crossentropy')
            model.fit(X_seq, y_seq, epochs=5, batch_size=16, verbose=0)
            save_model(model, f"models/lstm_model_{pod_name}.h5")

            print(f"[SUCCESS] Models retrained for {pod_name}")

        except Exception as e:
            print(f"[ERROR] Retraining failed for {svc.get('pod_name')}: {e}")
